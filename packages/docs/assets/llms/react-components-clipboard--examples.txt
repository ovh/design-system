# Clipboard

## Default

```tsx
<Clipboard value="Clipboard">
  <ClipboardControl />

  <ClipboardTrigger />
</Clipboard>
```

## Custom labels

```tsx
<Clipboard value="Custom labels">
  <ClipboardControl />

  <ClipboardTrigger
    labelCopy="Click to copy"
    labelCopySuccess="Successfully copied" />
</Clipboard>
```

## Disabled

```tsx
<Clipboard value="Disabled" disabled>
  <ClipboardControl />

  <ClipboardTrigger />
</Clipboard>
```

## Loading

```tsx
<Clipboard value="Loading" >
  <ClipboardControl loading />

  <ClipboardTrigger />
</Clipboard>
```

## Masked

```tsx
<Clipboard value="Masked" >
  <ClipboardControl maskOption={{ enable: true }} />

  <ClipboardTrigger />
</Clipboard>
```

## Accessibility - Form field

```tsx
<FormField>
  <FormFieldLabel>
    API key:
  </FormFieldLabel>

  <Clipboard value="loremipsum">
    <ClipboardControl />

    <ClipboardTrigger />
  </Clipboard>
</FormField>
```
