# File Thumbnail

## Default

```tsx
{
    const fakeFile = new File(['foo'], 'foo.txt', { type: 'text/plain' });

    return (
      <FileThumbnail file={ fakeFile } />
    );
  }
```

## Disabled

```tsx
{
    const fakeFile = new File(['foo'], 'foo.txt', { type: 'text/plain' });

    return (
      <FileThumbnail
        disabled
        file={ fakeFile } />
    );
  }
```

## Non Dismissible

```tsx
{
    const fakeFile = new File(['foo'], 'non-dismissible-file.txt', { type: 'text/plain' });

    return (
      <FileThumbnail
        dismissible={ false }
        file={ fakeFile } />
    );
  }
```

## Error

```tsx
{
    const fakeFile = new File(['foo'], 'foo.txt', { type: 'text/plain' });

    return (
      <FileThumbnail
        error="Something went wrong"
        file={ fakeFile } />
    );
  }
```

## Progress

```tsx
{
    const fakeFile = new File(['foo'], 'foo.txt', { type: 'text/plain' });

    return (
      <FileThumbnail
        file={ fakeFile }
        progress={ 45 }/>
    );
  }
```
