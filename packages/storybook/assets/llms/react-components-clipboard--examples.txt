# Clipboard

## Default

---

```jsx
<Clipboard value="Clipboard">
  <ClipboardControl />
  <ClipboardTrigger />
</Clipboard>
```

## Disabled

---

```jsx
<Clipboard
  disabled
  value="Disabled"
>
  <ClipboardControl />
  <ClipboardTrigger />
</Clipboard>
```

## Loading

---

```jsx
<Clipboard value="Loading">
  <ClipboardControl loading />
  <ClipboardTrigger />
</Clipboard>
```

## Masked

---

```jsx
<Clipboard value="Masked">
  <ClipboardControl
    maskOption={{
      enable: true
    }}
   />
  <ClipboardTrigger />
</Clipboard>
```

## Custom labels

---

```jsx
<Clipboard value="Custom labels">
  <ClipboardControl />
  <ClipboardTrigger
    labelCopy="Click to copy"
    labelCopySuccess="Successfully copied"
  />
</Clipboard>
```
