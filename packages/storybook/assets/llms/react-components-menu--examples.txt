# Menu

## Default

---

```jsx
<Menu>
  <MenuTrigger asChild>
    <Button>
      Open menu    </Button>
  </MenuTrigger>
  <MenuContent>
    <MenuItem value="profile">
      Profile    </MenuItem>
    <MenuItem value="settings">
      Settings    </MenuItem>
    <MenuItem value="logout">
      Logout    </MenuItem>
  </MenuContent>
</Menu>
```

## Controlled

---

```jsx
const [isOpen, setIsOpen] = useState(false);
  function toggleMenu() {
    setIsOpen(isOpen => !isOpen);
  }
  return <>
      <Button onClick={toggleMenu}>
        Toggle menu      </Button>
      <Menu open={isOpen}>
        <MenuTrigger asChild>
          <Icon name={ICON_NAME.ellipsisVertical} />
        </MenuTrigger>
        <MenuContent>
          <MenuItem value='profile'>Profile</MenuItem>
          <MenuItem value='settings'>Settings</MenuItem>
          <MenuItem value='logout'>Logout</MenuItem>
        </MenuContent>
      </Menu>
    </>;
}
```

## Disabled

---

```jsx
<Menu>
  <MenuTrigger asChild>
    <Button>
      Open menu    </Button>
  </MenuTrigger>
  <MenuContent>
    <MenuItem value="profile">
      Profile    </MenuItem>
    <MenuItem
      disabled
      value="settings"
    >
      Settings    </MenuItem>
    <MenuItem value="logout">
      Logout    </MenuItem>
  </MenuContent>
</Menu>
```
